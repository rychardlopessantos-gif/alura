// Funcionalidade de Curtir/Descurtir const botoesAcao = document.querySelectorAll(".post-actions button"); botoesAcao.forEach(function(botao) { botao.addEventListener("click", function() { let contador = botao.querySelector("span"); contador.textContent = Number(contador.textContent) + 1; }); }); // Funcionalidade do Modo Escuro const btnTema = document.getElementById("toggle-theme"); btnTema.addEventListener("click", function() { document.body.classList.toggle("dark-mode"); // Altera o ícone do botão dependendo do modo ativo if (document.body.classList.contains("dark-mode")) { btnTema.textContent = "☀️"; } else { btnTema.textContent = "🌙"; } });
