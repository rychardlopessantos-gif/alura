// Funcionalidade dos botões de reação (curtir / não curtir) const botoesReacao = document.querySelectorAll(".botoes-reacao button"); botoesReacao.forEach(function(botao) { botao.addEventListener("click", function() { let contador = botao.querySelector("span"); contador.textContent = parseInt(contador.textContent) + 1; }); }); // Funcionalidade do Botão do Modo Escuro const btnTema = document.getElementById("btn-tema"); btnTema.addEventListener("click", function() { document.body.classList.toggle("dark-mode"); // Altera o ícone do botão conforme o modo ativo if (document.body.classList.contains("dark-mode")) { btnTema.textContent = "☀️"; } else { btnTema.textContent = "🌙"; } });
